def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=True,
                      allow_nan=False).encode('ascii')


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def read_pin(root, relative, pin):
    name = PurePosixPath(relative)
    if (name.is_absolute() or '..' in name.parts or str(name) != relative or '\\' in relative
            or ':' in relative or any(ord(value) < 32 for value in relative)):
        raise ValueError('staging immutable relative name')
    path = root / relative
    if path.is_symlink() or path.resolve(strict=True) != path or not path.is_file():
        raise ValueError('staging immutable input path')
    if type(pin['bytes']) is not int or not 0 < pin['bytes'] <= 134217728:
        raise ValueError('staging immutable input bound')
    with path.open('rb') as stream:
        raw = stream.read(pin['bytes'] + 1)
    if len(raw) != pin['bytes'] or sha(raw) != pin['sha256']:
        raise ValueError('staging immutable input identity')
    return raw


def write_new(path, raw):
    if path.exists() or path.is_symlink() or path.parent.resolve(strict=True) != path.parent:
        raise ValueError('staging fresh output path')
    with path.open('xb') as stream:
        if stream.write(raw) != len(raw):
            raise ValueError('staging incomplete closed output')


def copy_manifest(source, target, manifest):
    if target.exists() or target.is_symlink() or target.parent.resolve(strict=True) != target.parent:
        raise ValueError('staging fresh manifest destination')
    total = sum(pin['bytes'] for pin in manifest.values())
    if not 0 < total <= 134217728 or len(manifest) > 128:
        raise ValueError('staging finite copy bound')
    target.mkdir()
    for relative, pin in manifest.items():
        raw = read_pin(source, relative, pin)
        path = target / relative
        path.parent.mkdir(parents=True, exist_ok=True)
        write_new(path, raw)
        if read_pin(target, relative, pin) != raw:
            raise ValueError('staging manifest copy parity')
    return {'entries': manifest, 'bytes': total, 'manifest_sha256': sha(canonical(manifest))}


def cleanup_owned(root, parent, ownership):
    if (root.parent != parent or not root.name.startswith('eq084-stage-')
            or root.is_symlink() or root.resolve(strict=True) != root
            or (root / 'ownership.json').read_bytes() != ownership):
        raise ValueError('staging cleanup root authority')
    entries = []
    for directory, directories, files in os.walk(root, topdown=False, followlinks=False):
        for name in files + directories:
            path = Path(directory) / name
            observed = path.lstat()
            if (len(entries) >= 500000 or path.is_symlink()
                    or getattr(observed, 'st_file_attributes', 0) & 0x400
                    or path.resolve(strict=True) != path or not path.is_relative_to(root)):
                raise ValueError('staging no-follow cleanup failed; retain root')
            entries.append((path, stat.S_ISDIR(observed.st_mode)))
    for path, directory in entries:
        observed = path.lstat()
        if (path.is_symlink() or getattr(observed, 'st_file_attributes', 0) & 0x400
                or path.resolve(strict=True) != path or not path.is_relative_to(root)):
            raise ValueError('staging cleanup drift; explicit failure')
        if directory:path.rmdir()
        else:path.unlink()
    root.rmdir()
    if root.exists():raise ValueError('staging final owned cleanup unfinished')
    return {'owned_nodes_removed': len(entries) + 1, 'owned_root_removed': True,
            'no_follow_checks': True}
