def record(path):
 with zipfile.ZipFile(path) as archive:
  entries=archive.infolist();assert len({v.filename for v in entries})==len(entries)
  assert sum(v.file_size for v in entries)<=268435456
  for item in entries:
   name=item.filename[:-1] if item.is_dir() else item.filename
   posix=PurePosixPath(name);windows=PureWindowsPath(name)
   assert name and ':' not in name and '\\' not in name and not posix.is_absolute() and not windows.drive and not windows.root
   assert '..' not in posix.parts and '/'.join(posix.parts)==name and item.file_size<=67108864
  names={v.filename for v in entries if not v.is_dir()}
  records=[v for v in names if v.endswith('.dist-info/RECORD')];assert len(records)==1
  assert archive.getinfo(records[0]).file_size<=1048576
  rows=list(csv.reader(io.StringIO(archive.read(records[0]).decode('utf-8'))))
  assert len({v[0] for v in rows})==len(rows) and {v[0] for v in rows}==names
  for name,digest,size in rows:
   if name==records[0]:assert digest==size=='';continue
   value=hashlib.sha256();count=0
   with archive.open(name) as stream:
    while chunk:=stream.read(1048576):value.update(chunk);count+=len(chunk)
   assert count==int(size) and digest=='sha256='+base64.urlsafe_b64encode(value.digest()).rstrip(b'=').decode('ascii')
