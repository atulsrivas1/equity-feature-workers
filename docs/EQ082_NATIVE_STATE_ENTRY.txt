import sys,time,json,threading
from pathlib import Path
sys.path.insert(0,sys.argv[1])
from test_jobs import setup
from equity_feature_service._audit import OwnedAudit
from equity_feature_service.loopback import qualification_server
mode=sys.argv[5]
if mode not in ('cancel','failure'):raise ValueError('owned state mode')
ready,stop,release=(Path(v) for v in sys.argv[2:5])
gate=threading.Event()
def transform(value):
 if mode=='failure':raise RuntimeError('owned private native callback text')
 return value
actions=('calculate','job_manage','retain','discover','raw_read','derived_read','export')
scheduler,service,tokens,clock,facts=setup(gate=gate if mode=='cancel' else None,cooperative=False,transform=transform,actions=actions,audit_store=OwnedAudit(owned_synthetic=True,valid_from_ns=0,expires_at_ns=300000000000))
service.ledger.credentials=tuple(type(c).provision(c.principal,('A' if n==0 else 'B')*43,0,3600000000000) for n,c in enumerate(service.ledger.credentials))

 requests+=1
 return service(environ,start_response)
server=qualification_server(service);server.set_app(app);server.timeout=.05
try:

 temporary=ready.with_name(ready.name+'.pending');temporary.write_bytes(data);temporary.replace(ready);ready.with_name(ready.name+'.complete').mkdir()
 deadline=time.monotonic()+20
 while requests<64 and not stop.exists() and time.monotonic()<deadline:
  if release.is_dir():gate.set()
  server.handle_request()
finally:
 gate.set();server.server_close()
 if scheduler.close(2) is not True:raise ValueError('owned native exit unconfirmed')
print(json.dumps({'marker':'owned-state-http-exited','mode':mode,'reads':facts['reads']},sort_keys=True,separators=(',',':')),flush=True)
