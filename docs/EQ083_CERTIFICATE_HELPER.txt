"""Uninvoked trusted qualification source; new synthetic private cert directory only."""
from datetime import datetime, timedelta, timezone
import hashlib
import ipaddress
import json
import os
from pathlib import Path
import sys
import importlib.metadata


def create_certificates(directory):
    if not sys.flags.isolated or sys.platform not in ('win32', 'linux'):
        raise ValueError('owned supported isolated certificate helper')
    if importlib.metadata.version('cryptography') != '50.0.2':
        raise ValueError('owned cryptography version')
    if type(directory) is not Path and not isinstance(directory, Path):
        raise ValueError('owned directory type')
    if not directory.is_absolute() or directory.exists() or directory.is_symlink():
        raise ValueError('new owned certificate directory required')
    parent = directory.parent
    if not parent.is_dir() or parent.is_symlink():
        raise ValueError('owned certificate parent')
    from cryptography import x509
    from cryptography.hazmat.primitives import hashes, serialization
    from cryptography.hazmat.primitives.asymmetric import rsa
    from cryptography.x509.oid import NameOID
    # Native crypto imports are from the pinned installed qualification environment.
    import cryptography
    if 'site-packages' not in Path(cryptography.__file__).resolve().parts:
        raise ValueError('owned installed crypto origin')
    directory.mkdir(mode=0o700)
    if os.name != 'nt':
        directory.chmod(0o700)
    now = datetime.now(timezone.utc)
    public = {}
    for label, address in (('valid', '127.0.0.1'), ('wrong-san', '127.0.0.2')):
        key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        name = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, 'EQ083 synthetic qualification')])
        certificate = (x509.CertificateBuilder().subject_name(name).issuer_name(name)
            .public_key(key.public_key()).serial_number(x509.random_serial_number())
            .not_valid_before(now - timedelta(days=1)).not_valid_after(now + timedelta(days=14))
            .add_extension(x509.SubjectAlternativeName([x509.IPAddress(ipaddress.ip_address(address))]), critical=False)
            .add_extension(x509.BasicConstraints(ca=False, path_length=None), critical=True)
            .add_extension(x509.ExtendedKeyUsage([x509.oid.ExtendedKeyUsageOID.SERVER_AUTH]), critical=False)
            .sign(key, hashes.SHA256()))
        cert_bytes = certificate.public_bytes(serialization.Encoding.PEM)
        key_bytes = key.private_bytes(serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8,
                                     serialization.NoEncryption())
        if not 0 < len(cert_bytes) <= 8192 or not 0 < len(key_bytes) <= 8192:
            raise ValueError('owned certificate byte bound')
        for suffix, data in (('cert.pem', cert_bytes), ('key.pem', key_bytes)):
            path = directory / (label + '-' + suffix)
            descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
            stream = None
            try:
                stream = os.fdopen(descriptor, 'wb')
                descriptor = None
                if stream.write(data) != len(data):
                    raise ValueError('owned certificate short write')
                stream.flush()
            finally:
                original = sys.exc_info()[0] is not None
                try:
                    if stream is not None:
                        stream.close()
                    elif descriptor is not None:
                        os.close(descriptor)
                except BaseException:
                    if not original:
                        raise
        # Do not print, return or archive private-key bytes or their file contents.
        public[label] = dict(certificate=label + '-cert.pem', sha256=hashlib.sha256(cert_bytes).hexdigest(),
                             san_ip=address, expires=certificate.not_valid_after_utc.isoformat())
    # Failure leaves this newly owned directory quarantined; it is never reused as complete.
    return public


if __name__ == '__main__':
    if len(sys.argv) != 2:
        raise ValueError('owned certificate arguments')
    receipt = create_certificates(Path(sys.argv[1]))
    print(json.dumps(receipt, sort_keys=True, separators=(',', ':')))
