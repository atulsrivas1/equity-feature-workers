def require(ok, reason):
    if not ok:
        raise ValueError(reason)

def tick(start):
    require(time.monotonic() - start <= 180, 'cooperative deadline')

def member_path(name):
    require(type(name) is str and 0 < len(name) <= 1024, 'member name bound')
    require(not any(ord(c) < 32 or ord(c) == 127 for c in name), 'member control character')
    require('\\' not in name and ':' not in name and not name.startswith('/'), 'member path syntax')
    parts = name.rstrip('/').split('/')
    require(all(p not in ('', '.', '..') for p in parts), 'member traversal')
    return name

def tags_for(filename):
    stem = filename[:-4]
    py, abi, platform = stem.rsplit('-', 3)[-3:]
    return sorted('-'.join(x) for x in itertools.product(py.split('.'), abi.split('.'), platform.split('.')))

def headers(data):
    require(len(data) <= 1048576, 'metadata bound')
    text = data.decode('utf-8', 'strict')
    message = email.parser.Parser(policy=email.policy.default).parsestr(text)
    require(not message.defects, 'metadata parse defects')
    return message

def one(message, name):
    values = message.get_all(name, [])
    require(len(values) == 1, 'metadata singleton ' + name)
    return str(values[0])

def normalize(name):
    return re.sub('[-_.]+', '-', name).lower()
