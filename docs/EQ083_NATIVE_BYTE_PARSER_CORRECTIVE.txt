def require(ok, reason):
    if not ok:
        raise ValueError(reason)

def view(data, offset, size):
    require(type(data) is bytes and type(offset) is int and type(size) is int, 'byte view types')
    require(offset >= 0 and size >= 0 and offset <= len(data) and size <= len(data) - offset, 'byte range')
    return data[offset:offset + size]

def unpack(data, offset, format):
    return struct.unpack(format, view(data, offset, struct.calcsize(format)))

def text_at(data, offset, limit):
    require(type(limit) is int and 0 < limit <= 4096, 'string bound')
    part = view(data, offset, min(limit, len(data) - offset))
    end = part.find(b'\0')
    require(end >= 0, 'string terminator')
    result = part[:end].decode('ascii', 'strict')
    require(result and all(32 <= ord(c) < 127 for c in result), 'string characters')
    return result

def manifest_fields(raw):
    require(0 < len(raw) <= 65536, 'manifest byte bound')
    # Preserve byte identity; reject DTD/entity declarations before XML parsing.
    text = raw.decode('utf-8-sig', 'strict').rstrip('\0')
    require('<!DOCTYPE' not in text.upper() and '<!ENTITY' not in text.upper(), 'manifest DTD')
    root = ET.fromstring(text)
    elements = list(root.iter())
    require(len(elements) <= 1024, 'manifest element bound')
    return dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest(),
                elements=[dict(tag=n.tag, attributes=dict(n.attrib), text=n.text, tail=n.tail) for n in elements])

def pe(data):
    require(view(data, 0, 2) == b'MZ', 'DOS signature')
    header, = unpack(data, 60, '<I')
    require(view(data, header, 4) == b'PE\0\0', 'PE signature')
    machine, count, timestamp, symbol_pointer, symbols, opt_size, flags = unpack(data, header + 4, '<HHIIIHH')
    require(machine in (0x14c, 0x8664, 0xaa64) and 0 < count <= 96, 'PE machine sections')
    optional = header + 24
    magic, = unpack(data, optional, '<H')
    require(magic in (0x10b, 0x20b), 'PE optional format')
    width = 4 if magic == 0x10b else 8
    number_offset = 92 if width == 4 else 108
    directory_offset = number_offset + 4
    require(directory_offset <= opt_size <= 4096, 'PE optional size')
    opt = view(data, optional, opt_size)
    image_base, = unpack(opt, 28 if width == 4 else 24, '<I' if width == 4 else '<Q')
    image_size, header_size = unpack(opt, 56, '<II')
    entry, = unpack(opt, 16, '<I')
    subsystem, dll_flags = unpack(opt, 68, '<HH')
    number, = unpack(opt, number_offset, '<I')
    require(0 <= number <= 16 and directory_offset + number * 8 <= opt_size, 'PE directories')
    require(0 < header_size <= len(data) and image_size >= header_size, 'PE image/header size')
    directories = [dict(index=i, address=unpack(opt, directory_offset + i * 8, '<II')[0],
                        size=unpack(opt, directory_offset + i * 8, '<II')[1]) for i in range(number)]
    sections = []
    table = optional + opt_size
    require(table + count * 40 <= header_size, 'PE section table')
    for i in range(count):
        row = view(data, table + i * 40, 40)
        virtual_size, rva, raw_size, raw_offset, reloc, line_ptr, reloc_count, line_count, characteristics = unpack(row, 8, '<IIIIIIHHI')
        view(data, raw_offset, raw_size)
        require(rva + max(virtual_size, raw_size) <= image_size, 'PE section virtual bound')
        sections.append(dict(name_hex=row[:8].hex(), virtual_size=virtual_size, rva=rva, raw_size=raw_size,
                             raw_offset=raw_offset, relocation_pointer=reloc, line_pointer=line_ptr,
                             relocation_count=reloc_count, line_count=line_count, characteristics=characteristics))
    def rva_view(address, length):
        require(type(address) is int and type(length) is int and address >= 0 and length > 0, 'PE RVA types')
        choices = []
        if address + length <= header_size:
            choices.append(address)
        for section in sections:
            delta = address - section['rva']
            if delta >= 0 and delta + length <= section['raw_size']:
                choices.append(section['raw_offset'] + delta)
        require(len(choices) == 1, 'PE unique file-backed RVA')
        return view(data, choices[0], length)
    def rva_text(address):
        chars = bytearray()
        for i in range(4096):
            byte = rva_view(address + i, 1)[0]
            if byte == 0:
                result = bytes(chars).decode('ascii', 'strict')
                require(result and all(32 <= ord(c) < 127 for c in result), 'PE import string')
                return result
            chars.append(byte)
        raise ValueError('PE import string bound')
    def directory(index):
        if index >= len(directories):
            return 0, 0
        row = directories[index]
        require((row['address'] == 0) == (row['size'] == 0), 'PE directory zero pair')
        return row['address'], row['size']
    def thunk_names(address):
        result = []
        for i in range(4096):
            value, = struct.unpack('<I' if width == 4 else '<Q', rva_view(address + i * width, width))
            if value == 0:
                return result
            if value & (1 << (width * 8 - 1)):
                require(value & ~((1 << (width * 8 - 1)) | 65535) == 0, 'PE ordinal reserved bits')
                result.append(dict(ordinal=value & 65535))
            else:
                hint, = struct.unpack('<H', rva_view(value, 2))
                result.append(dict(hint=hint, name=rva_text(value + 2)))
        raise ValueError('PE thunk count')
    imports = []
    for index, row_size, delayed in ((1, 20, False), (13, 32, True)):
        address, size = directory(index)
        if address == 0:
            continue
        require(size <= 1048576, 'PE imports size')
        terminated = False
        for offset in range(0, size - row_size + 1, row_size):
            require(len(imports) < 1024, 'PE imports count')
            values = struct.unpack('<IIIII' if not delayed else '<IIIIIIII', rva_view(address + offset, row_size))
            if not any(values):
                terminated = True
                break
            if delayed:
                attributes, name, handle, iat, lookup, bound, unload, stamp = values
                require(attributes in (0, 1), 'PE delay addressing')
                def convert(v):
                    if attributes == 1 or v == 0:
                        return v
                    require(v >= image_base, 'PE delay VA base')
                    return v - image_base
                name, lookup, iat = convert(name), convert(lookup), convert(iat)
            else:
                lookup, stamp, forward, name, iat = values
                attributes = None
            require(name > 0 and (lookup or iat) > 0, 'PE import descriptor')
            imports.append(dict(delayed=delayed, raw_descriptor=list(values), library=rva_text(name),
                                symbols=thunk_names(lookup or iat)))
        require(terminated, 'PE import descriptor terminator')
    resources = []
    resource_rva, resource_size = directory(2)
    if resource_rva:
        require(resource_size <= 16777216, 'PE resource bounds')
        visited = set()
        def resource_walk(relative, path):
            require(relative not in visited and len(visited) < 4096 and len(path) <= 8, 'PE resource cycle depth')
            visited.add(relative)
            require(0 <= relative and relative + 16 <= resource_size, 'PE resource directory range')
            fields = struct.unpack('<IIHHHH', rva_view(resource_rva + relative, 16))
            entries = fields[4] + fields[5]
            require(entries <= 4096 and relative + 16 + entries * 8 <= resource_size, 'PE resource entries')
            for i in range(entries):
                key, target = struct.unpack('<II', rva_view(resource_rva + relative + 16 + i * 8, 8))
                if key & 0x80000000:
                    location = key & 0x7fffffff
                    require(location + 2 <= resource_size, 'PE resource name range')
                    length, = struct.unpack('<H', rva_view(resource_rva + location, 2))
                    require(length <= 1024 and location + 2 + length * 2 <= resource_size, 'PE resource name length')
                    label = rva_view(resource_rva + location + 2, length * 2).decode('utf-16-le', 'strict')
                else:
                    label = key
                if target & 0x80000000:
                    resource_walk(target & 0x7fffffff, path + [label])
                else:
                    require(target + 16 <= resource_size and len(resources) < 4096, 'PE resource data entry')
                    payload_rva, length, codepage, reserved = struct.unpack('<IIII', rva_view(resource_rva + target, 16))
                    require(reserved == 0 and length <= 16777216, 'PE resource data shape')
                    payload = rva_view(payload_rva, length) if length else b''
                    row = dict(path=path + [label], bytes=length, sha256=hashlib.sha256(payload).hexdigest(), codepage=codepage)
                    if row['path'][0] == 24:
                        row['manifest'] = manifest_fields(payload)
                    resources.append(row)
        resource_walk(0, [])
    certificate = None
    cert_offset, cert_size = directory(4)  # Security directory is a file offset, not RVA.
    if cert_offset:
        certificate = dict(file_offset=cert_offset, bytes=cert_size, sha256=hashlib.sha256(view(data, cert_offset, cert_size)).hexdigest(), signature_verified=False)
    return dict(format='PE', machine=machine, bits=width * 8, coff=dict(sections=count, timestamp=timestamp,
                symbol_pointer=symbol_pointer, symbols=symbols, characteristics=flags), entry_rva=entry,
                image_base=image_base, image_size=image_size, header_size=header_size, subsystem=subsystem,
                dll_characteristics=dll_flags, optional_bytes=opt_size, optional_sha256=hashlib.sha256(opt).hexdigest(), optional_raw_hex=opt.hex(),
                directories=directories, sections=sections, imports=imports, resources=resources, certificate=certificate,
                runtime_dynamic_loading_resolved=False, api_set_sxs_host_authority=False)

def elf(data):
    require(view(data, 0, 7) == b'\x7fELF\x02\x01\x01', 'ELF64 little endian version')
    kind, machine, version, entry, phoff, shoff, flags, ehsize, phsize, phnum, shsize, shnum, shstr = unpack(data, 16, '<HHIQQQIHHHHHH')
    require(kind in (2, 3) and machine == 62 and version == 1 and ehsize == 64, 'ELF header')
    require(phsize == 56 and 0 < phnum <= 1024 and shsize == 64 and 0 < shnum <= 4096 and shstr < shnum, 'ELF tables; extended counts unsupported')
    programs = []
    for i in range(phnum):
        ptype, pflags, offset, vaddr, physical, filesz, memsz, align = unpack(data, phoff + i * phsize, '<IIQQQQQQ')
        view(data, offset, filesz)
        require(filesz <= memsz, 'ELF segment file-memory size')
        programs.append(dict(type=ptype, flags=pflags, offset=offset, virtual=vaddr, physical=physical, file_bytes=filesz, memory_bytes=memsz, alignment=align))
    sections = []
    for i in range(shnum):
        name, stype, sflags, address, offset, size, link, info, align, entsize = unpack(data, shoff + i * shsize, '<IIQQQQIIQQ')
        if stype != 8:  # SHT_NOBITS has no file payload.
            view(data, offset, size)
        require(link < shnum, 'ELF section link')
        sections.append(dict(name_index=name, type=stype, flags=sflags, address=address, offset=offset, bytes=size, link=link, info=info, alignment=align, entry_bytes=entsize))
    strings = sections[shstr]
    require(strings['type'] == 3, 'ELF section name string table')
    string_data = view(data, strings['offset'], strings['bytes'])
    for row in sections:
        row['name'] = '' if row['name_index'] == 0 else text_at(string_data, row['name_index'], 4096)
    def virtual_offset(address, length):
        choices = []
        for row in programs:
            delta = address - row['virtual']
            if row['type'] == 1 and delta >= 0 and delta + length <= row['file_bytes']:
                choices.append(row['offset'] + delta)
        require(len(choices) == 1 and length > 0, 'ELF unique file-backed virtual range')
        return choices[0]
    def virtual_view(address, length):
        return view(data, virtual_offset(address, length), length)
    dynamic_programs = [p for p in programs if p['type'] == 2]
    require(len(dynamic_programs) == 1, 'ELF one dynamic segment')
    dyn = dynamic_programs[0]
    require(dyn['file_bytes'] % 16 == 0 and dyn['file_bytes'] <= 1048576, 'ELF dynamic bounds')
    dynamic = []
    terminated = False
    for offset in range(0, dyn['file_bytes'], 16):
        tag, value = unpack(data, dyn['offset'] + offset, '<qQ')
        if tag == 0:
            terminated = True
            break
        dynamic.append(dict(tag=tag, value=value))
    require(terminated, 'ELF dynamic terminator')
    def single(tag, optional=False):
        values = [r['value'] for r in dynamic if r['tag'] == tag]
        require(len(values) == 1 or optional and not values, 'ELF dynamic singleton')
        return values[0] if values else None
    table_rva = single(5)
    table_bytes = single(10)
    require(0 < table_bytes <= 16777216, 'ELF dynamic string bounds')
    table_data = virtual_view(table_rva, table_bytes)
    declarations = [dict(tag=r['tag'], value=text_at(table_data, r['value'], 4096)) for r in dynamic if r['tag'] in (1, 14, 15, 29)]
    versions = []
    version_indexes = {}
    version_rva = single(0x6ffffffe, True)
    version_count = single(0x6fffffff, True)
    require((version_rva is None) == (version_count is None), 'ELF version count pair')
    if version_rva is not None:
        require(0 < version_count <= 512, 'ELF version count')
        cursor = version_rva
        visited = set()
        for i in range(version_count):
            require(cursor not in visited, 'ELF version cycle')
            visited.add(cursor)
            revision, count, file_index, auxiliary, next_offset = struct.unpack('<HHIII', virtual_view(cursor, 16))
            require(revision == 1 and 0 < count <= 512 and auxiliary > 0, 'ELF version requirement')
            aux_cursor = cursor + auxiliary
            aux_seen = set()
            names = []
            for j in range(count):
                require(aux_cursor not in aux_seen, 'ELF version auxiliary cycle')
                aux_seen.add(aux_cursor)
                hash_value, aflags, other, name, following = struct.unpack('<IHHII', virtual_view(aux_cursor, 16))
                require(2 <= other <= 32767 and other not in version_indexes, 'ELF supported unique version index')
                version_name = text_at(table_data, name, 4096)
                version_indexes[other] = dict(file=text_at(table_data, file_index, 4096), name=version_name)
                names.append(dict(hash=hash_value, flags=aflags, other=other, name=version_name))
                require((following == 0) == (j == count - 1), 'ELF version auxiliary terminator')
                aux_cursor += following
            versions.append(dict(file=text_at(table_data, file_index, 4096), versions=names))
            require((next_offset == 0) == (i == version_count - 1), 'ELF version next terminator')
            cursor += next_offset
    undefined = []
    symbol_tables = [r for r in sections if r['type'] == 11]
    require(len(symbol_tables) == 1, 'ELF one dynamic symbol section')
    symbols = symbol_tables[0]
    require(symbols['entry_bytes'] == 24 and symbols['bytes'] % 24 == 0 and symbols['bytes'] // 24 <= 65536, 'ELF dynamic symbols bounds')
    symstrings = sections[symbols['link']]
    require(symstrings['type'] == 3, 'ELF symbol string section')
    symtab_rva = single(6)
    syment = single(11)
    require(syment == 24 and symbols['address'] == symtab_rva and symbols['offset'] == virtual_offset(symtab_rva, symbols['bytes']), 'ELF section/dynamic symbol reconciliation')
    require(symstrings['address'] == table_rva and symstrings['offset'] == virtual_offset(table_rva, table_bytes) and symstrings['bytes'] == table_bytes, 'ELF section/dynamic string reconciliation')
    symdata = table_data
    symbol_count = symbols['bytes'] // 24
    versym_rva = single(0x6ffffff0, True)
    versym_data = virtual_view(versym_rva, symbol_count * 2) if versym_rva is not None and symbol_count else None
    for offset in range(0, symbols['bytes'], 24):
        name, info, other, index, value, size = unpack(data, symbols['offset'] + offset, '<IBBHQQ')
        if index == 0 and name:
            version_index = struct.unpack('<H', versym_data[(offset // 24) * 2:(offset // 24) * 2 + 2])[0] if versym_data is not None else None
            require(version_index is None or (version_index & 32767) in (0, 1) or (version_index & 32767) in version_indexes, 'ELF undefined symbol version reconciliation')
            undefined.append(dict(name=text_at(symdata, name, 4096), info=info, other=other, value=value, size=size, version_index=version_index,
                                  version_requirement=version_indexes.get(version_index & 32767) if version_index is not None else None))
    return dict(format='ELF64', machine=machine, kind=kind, entry=entry, flags=flags, osabi=data[7], abi_version=data[8],
                header=dict(phoff=phoff, shoff=shoff, ehsize=ehsize, phsize=phsize, phnum=phnum, shsize=shsize, shnum=shnum, shstr=shstr, raw_hex=view(data, 0, 64).hex()),
                programs=programs, sections=sections, dynamic=dynamic, declarations=declarations,
                version_requirements=versions, undefined_symbols=undefined,
                runtime_resolution_proven=False, dynamic_loading_complete=False)

def close_preserving(resource, primary):
    try:
        resource.close()
    except BaseException as cleanup:
        if primary is not None:
            raise primary from cleanup
        raise

def read_original(path, size, digest, start):
    stream = path.open('rb')
    try:
        chunks = []
        count = 0
        while True:
            require(time.monotonic() - start <= 180, 'cooperative deadline')
            chunk = stream.read(min(65536, size - count + 1))
            if not chunk:
                break
            count += len(chunk)
            require(count <= size, 'original archive growth')
            chunks.append(chunk)
        raw = b''.join(chunks)
        require(count == size and hashlib.sha256(raw).hexdigest() == digest, 'original archive bytes')
        return raw
    finally:
        close_preserving(stream, sys.exception())

def deadline(start):
    require(time.monotonic() - start <= 180, 'cooperative deadline')

def main():
    require(len(sys.argv) == 1 and sys.platform == 'win32' and sys.version_info[:3] == (3, 12, 10), 'fixed invocation')
    start = time.monotonic()
    parent = Path(__file__).absolute().parents[3]
    require(parent.name == 'equity-r6-postreview-20261008', 'fixed source layout')
    records = []
    read_bytes = 0
    for artifact in MANIFEST:
        require(0 < artifact['bytes'] <= 67108864, 'archive bounds')
        raw = read_original(parent / artifact['root'] / artifact['filename'], artifact['bytes'], artifact['sha256'], start)
        read_bytes += len(raw)
        require(read_bytes <= 22978120, 'aggregate archive bytes')
        archive = zipfile.ZipFile(io.BytesIO(raw))
        try:
            for member, expected in artifact['native_members'].items():
                require(time.monotonic() - start <= 180, 'cooperative deadline')
                info = archive.getinfo(member)
                require(info.orig_filename == info.filename == member and info.file_size == expected['bytes'] <= 67108864, 'native original member')
                data = archive.read(info)
                require(len(data) == expected['bytes'] and hashlib.sha256(data).hexdigest() == expected['sha256'], 'native member bytes')
                row = dict(archive=artifact['filename'], member=member, bytes=len(data), sha256=expected['sha256'], admitted=False)
                try:
                    row['declarations'] = pe(data) if expected['native_magic'] == 'PE' else elf(data)
                    row['parsed'] = True
                except (ValueError, struct.error, UnicodeError, ET.ParseError) as error:
                    row['parsed'] = False
                    row['error_type'] = type(error).__name__
                    row['error'] = str(error)
                deadline(start)  # Outside retained format errors: expiry fails whole operation.
                records.append(row)
        finally:
            close_preserving(archive, sys.exception())
    deadline(start)
    require(len(records) == 424 and read_bytes == 22978120, 'complete native selection')
    result = dict(schema='eq083.static-native-byte-declarations.v1', records=records,
                  parsed=sum(r['parsed'] for r in records), unresolved=sum(not r['parsed'] for r in records),
                  read_bytes=read_bytes, admitted=False, host_import_authority=False, runtime_resolver_complete=False,
                  security_disposition='unassessed')
    body = json.dumps(result, sort_keys=True, ensure_ascii=True, separators=(',', ':')).encode('ascii') + b'\n'
    deadline(start)
    require(len(body) <= 8388608, 'native declaration output bound')
    written = sys.stdout.buffer.write(body)
    require(type(written) is int and written == len(body), 'native declaration output write')
    deadline(start)
    sys.stdout.buffer.flush()
    deadline(start)

if __name__ == '__main__':
